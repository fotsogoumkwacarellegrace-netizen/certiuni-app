=== model Payment ===

id String @id @default(uuid()) @db.Uuid
requestId String  @db.Uuid
request   Request @relation(fields: [requestId], references: [id], onDelete: Cascade)
certificateId String?      @db.Uuid
certificate   Certificate? @relation(fields: [certificateId], references: [id], onDelete: SetNull)
studentId String?  @db.Uuid
student   Student? @relation(fields: [studentId], references: [id], onDelete: SetNull)
amount               Decimal       @db.Decimal(12, 2)
currency             String        @default("XAF") @db.VarChar(10)
provider             PaymentMethod
status               PaymentStatus @default(PENDING)
transactionId        String?       @unique @db.VarChar(255)
transactionReference String?       @unique @db.VarChar(255)
createdAt DateTime @default(now())
updatedAt DateTime @updatedAt
@@index([requestId])
@@index([certificateId])
@@index([studentId])
@@map("payments")

=== model Request ===

@@index([studentId])
@@index([serviceId])
@@index([status])
@@map("requests")
